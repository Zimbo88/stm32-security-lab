ENTRY(Reset_Handler)

MEMORY
{
    FLASH (rx)  : ORIGIN = 0x08000000, LENGTH = 32K
    RAM   (rwx) : ORIGIN = 0x20000000, LENGTH = 128K
}

_estack = ORIGIN(RAM) + LENGTH(RAM);

_flash_region_start = ORIGIN(FLASH);
_flash_region_end = ORIGIN(FLASH) + LENGTH(FLASH);

SECTIONS
{
    .isr_vector :
    {
        . = ALIGN(4);
        KEEP(*(.isr_vector))
        . = ALIGN(4);
    } > FLASH

    .text :
    {
        . = ALIGN(4);
        *(.text*)
        *(.rodata*)
        . = ALIGN(4);
        _etext = .;
    } > FLASH

    .data : AT(_etext)
    {
        . = ALIGN(4);
        _sdata = .;
        *(.data*)
        . = ALIGN(4);
        _edata = .;
    } > RAM

    .bss (NOLOAD) :
    {
        . = ALIGN(8);
        _sbss = .;
        *(.bss*)
        *(COMMON)
        . = ALIGN(8);
        _ebss = .;
    } > RAM

    /*
     * Include the initial values of .data because they are stored in flash
     * even though the section executes from RAM.
     */
    _flash_image_end = LOADADDR(.data) + SIZEOF(.data);

    ASSERT(
        _flash_image_end <= _flash_region_end,
        "ERROR: bootloader flash image exceeds reserved 32 KiB region"
    )
}

    29	verify_status_t signed_image_verify(void)
    30	{
    31	    const signed_manifest_t *m = manifest();
    32
    33	    if (sizeof(signed_manifest_t) != SIGNED_MANIFEST_SIZE) {
    34	        return VERIFY_BAD_HEADER_VERSION;
    35	    }
    36
    37	    if (m->magic != SIGNED_IMAGE_MAGIC) {
    38	        return VERIFY_BAD_MAGIC;
    39	    }
    40
    41	    if (m->header_version != SIGNED_HEADER_VERSION) {
    42	        return VERIFY_BAD_HEADER_VERSION;
    43	    }
    44
    45	    if (m->image_version < MIN_IMAGE_VERSION) {
    46	        return VERIFY_ROLLBACK_VERSION;
    47	    }
    48
    49	    if (m->vector_address != APPLICATION_BASE_ADDRESS) {
    50	        return VERIFY_BAD_VECTOR_ADDRESS;
    51	    }
    52
    53	    if ((m->image_size < 8U) || (m->image_size > MAX_PAYLOAD_SIZE)) {
    54	        return VERIFY_BAD_SIZE;
    55	    }
    56
    57	    const uint32_t msp =
    58	        *(volatile const uint32_t *)(m->vector_address + 0U);
    59	    const uint32_t reset =
    60	        *(volatile const uint32_t *)(m->vector_address + 4U);
    61
    62	    if ((msp < BOARD_SRAM_BASE) || (msp > (BOARD_SRAM_BASE + BOARD_SRAM_SIZE))) {
    63	        return VERIFY_BAD_STACK;
    64	    }
    65
    66	    if (((reset & 1UL) == 0U) ||
    67	        ((reset & ~1UL) < m->vector_address) ||
    68	        ((reset & ~1UL) >= (m->vector_address + m->image_size))) {
    69	        return VERIFY_BAD_RESET_VECTOR;
    70	    }

    74	        computed_hash,
    75	        (const uint8_t *)m->vector_address,
    76	        (size_t)m->image_size
    77	    );

    80	        crypto_wipe(computed_hash, sizeof(computed_hash));
    81	        return VERIFY_BAD_PAYLOAD_HASH;
    82	    }

    92	    }
